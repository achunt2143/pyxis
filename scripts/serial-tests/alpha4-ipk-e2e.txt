# Alpha 4 acceptance test — a third-party .ipk, end to end.
#
# Run with:
#   scripts/serial-test.py \
#     --push <pkg>.ipk:/var/lib/pyxis/incoming/<pkg>.ipk \
#     --script testing/alpha4-ipk-e2e.txt
#
# The flow this proves (ALPHA_ROADMAP.md alpha 4 item B): build a package with
# pyxis-sdk, discover it, install it through the *installer service* (not raw
# opkg — the service path is what the Installer app uses), see Nova pick it up
# as an opkg-tier app, launch it, and remove it again.
#
# echo-cli needs the FULL destination name; the short form returns
# "Service not registered", which reads like the daemon is down.

# ---------- preconditions ----------
pgrep -a echo-daemon | head -1
pgrep -a evo | head -1
opkg list-installed | wc -l
ls /usr/pyxis/applications/ 2>&1 | head
echo-cli call org.geckoOS.nova List --interface org.geckoOS.nova --args '{}' | grep -o '"id": "[^"]*"' | sort

# ---------- discovery, before installing ----------
ls -l /var/lib/pyxis/incoming/
echo-cli call org.geckoOS.installer ListAvailable --interface org.geckoOS.installer --args '{}'
echo-cli call org.geckoOS.installer Inspect --interface org.geckoOS.installer --args '{"packagePath":"/var/lib/pyxis/incoming/com.pyxis.hello_0.1.0_all.ipk"}'

# ---------- install ----------
echo-cli call org.geckoOS.installer Install --interface org.geckoOS.installer --args '{"packagePath":"/var/lib/pyxis/incoming/com.pyxis.hello_0.1.0_all.ipk"}'

# ---------- did opkg actually place the files ----------
opkg list-installed
ls -la /usr/pyxis/applications/com.pyxis.hello/
cat /usr/pyxis/applications/com.pyxis.hello/appinfo.json

# ---------- did Nova index it ----------
echo-cli call org.geckoOS.nova List --interface org.geckoOS.nova --args '{}' | grep -o '"id": "[^"]*"' | sort
echo-cli call org.geckoOS.nova GetInfo --interface org.geckoOS.nova --args '{"appId":"com.pyxis.hello"}'

# ---------- the derived artefacts Nova owns ----------
ls -la /var/lib/pyxis/nova/native/
cat /var/lib/pyxis/nova/native/com.pyxis.hello/manifest.json 2>&1
grep com.pyxis.hello /run/pyxis/nova/origins.map 2>&1

# ---------- can AppHost see it ----------
echo-cli call org.geckoOS.appHost ListApps --interface org.geckoOS.appHost --args '{}' | grep -o '"id": "[^"]*"' | sort
pyxis-appctl list 2>&1 | head -20

# ---------- launch ----------
echo-cli call org.geckoOS.nova Launch --interface org.geckoOS.nova --args '{"appId":"com.pyxis.hello"}'
sleep 6
pgrep -af vela | head -5
echo-cli call org.geckoOS.appHost ListRunningApps --interface org.geckoOS.appHost --args '{}'
# Did the document actually load and run? The test page sets document.title
# from whether window.geckoOS attached, and Evo logs every title it is told.
# A mapped surface only proves a window exists; this proves the pyxis://
# origin served the file, the script ran, and whether a *third-party* app
# gets the bridge (Nova maps appinfo permissions deny-by-default, so a
# built-in proving the bridge says nothing about a packaged one).
grep -a 'Hello Pyxis' /var/log/evo.log | tail -3
grep -a 'com.pyxis.hello' /var/log/evo.log | grep -ai 'title\|commit\|map' | tail -5
grep -ai 'com.pyxis.hello' /var/log/vela.log | tail -5

# ---------- close ----------
echo-cli call org.geckoOS.nova Close --interface org.geckoOS.nova --args '{"appId":"com.pyxis.hello"}'

# ---------- the tier split: Nova must refuse an opkg-owned app ----------
echo-cli call org.geckoOS.nova Uninstall --interface org.geckoOS.nova --args '{"appId":"com.pyxis.hello"}'

# ---------- remove through the installer, which owns the package layer ----------
echo-cli call org.geckoOS.installer Remove --interface org.geckoOS.installer --args '{"package":"com.pyxis.hello"}'
opkg list-installed
ls /usr/pyxis/applications/ 2>&1 | head
echo-cli call org.geckoOS.nova List --interface org.geckoOS.nova --args '{}' | grep -o '"id": "[^"]*"' | sort
ls /var/lib/pyxis/nova/native/ 2>&1
