# Alpha 5: live window thumbnails (evo.GetWindowThumbnails, evo/src/shell/thumbnails.rs).
#
#   scripts/serial-test.py --memory 4096 --push target/release/evo:/usr/bin/evo \
#       --pull /tmp/thumb-files.b64:target/serial-shots/thumb-files.b64 \
#       --script scripts/serial-tests/alpha5-thumbnails.txt
#
# (A suite of its own because alpha5-animations is being edited by another
# lane; it follows the same conventions and needs the same push.) The echo-cli
# calls are root and need no vela rebuild; Bloom's use of the call does (the
# bridge entry), which dev-server screenshots cover instead.
#
# Asserts: one picture per app window, at the window's aspect ratio and inside
# the size bounds, each a real data:image/jpeg URL; a request for given ids
# gets exactly those; a minimized window still has its last picture; the call
# is inside its latency budget, cold and cached; bad arguments are refused.
# The latency lines (`LAT-*`) print milliseconds for five windows, which is
# what the report quotes.
#
# LOOK at the pulled picture: decode target/serial-shots/thumb-files.b64
# (`base64 -d thumb-files.b64 > thumb-files.jpg`) and view it: the Files window,
# title bar included, rounded corners black, at card size.

wid() { echo-cli call org.geckoOS.evo ListWindows --interface org.geckoOS.evo --args '{}' | awk -v app="$1" '/"app_id"/ { on = index($0, "\"" app "\"") > 0 } on && /"window_id":/ { gsub(/[",:]/, ""); print $2; exit }'; }
# "window_id width height" of every window ListWindows lists, one per line.
wins() { echo-cli call org.geckoOS.evo ListWindows --interface org.geckoOS.evo --args '{}' | awk '/"app_id"/ { app = $2; gsub(/[",]/, "", app) } /"height":/ { gsub(/[",:]/, ""); h = $2 } /"width":/ { gsub(/[",:]/, ""); w = $2 } /"window_id":/ { gsub(/[",:]/, ""); print $2, w, h, app }'; }
# "window_id width height datalen prefix-ok" per thumbnail.
thumbs() { echo-cli call org.geckoOS.evo GetWindowThumbnails --interface org.geckoOS.evo --args "$1" | awk '/"data":/ { len = length($0); ok = index($0, "data:image/jpeg;base64,") > 0 } /"height":/ { gsub(/[",:]/, ""); h = $2 } /"width":/ { gsub(/[",:]/, ""); w = $2 } /"window_id":/ { gsub(/[",:]/, ""); print $2, w, h, len, ok }'; }
ms() { echo $(( $(date +%s%N) / 1000000 )); }

echo "=== five app windows ==="
for A in files notes tasks settings terminal; do echo-cli call org.geckoOS.nova Launch --interface org.geckoOS.nova --args "{\"appId\":\"$A\"}" | grep -E 'sessionId|Error'; sleep 10; done
sleep 5
wins | grep -v bloom
#!must FIVE-APP-WINDOWS
test "$(wins | grep -v -e bloom -e vela-default | wc -l)" -ge 5 && echo FIVE-APP-WINDOWS || echo "APP-WINDOWS-$(wins | grep -v -e bloom -e vela-default | wc -l)"

echo "=== one picture per app window ==="
# Warm-up so the baseline below is echo-cli itself, not first-use costs.
echo-cli call org.geckoOS.evo GetOverlayState --interface org.geckoOS.evo --args '{}' > /dev/null
T0=$(ms); echo-cli call org.geckoOS.evo GetOverlayState --interface org.geckoOS.evo --args '{}' > /dev/null; T1=$(ms)
echo "LAT-BASELINE echo-cli round trip: $((T1 - T0)) ms"
T0=$(ms); thumbs '{"max_width":480,"max_height":300}' > /tmp/t1.txt; T1=$(ms)
echo "LAT-COLD five windows: $((T1 - T0)) ms (includes echo-cli)"
cat /tmp/t1.txt
#!must ONE-PER-WINDOW
test "$(wc -l < /tmp/t1.txt)" -ge 5 && echo ONE-PER-WINDOW || echo "THUMBS-$(wc -l < /tmp/t1.txt)"
#!must ALL-JPEG-DATA-URLS
awk '$5 != 1 || $4 < 500 { bad = 1 } END { print bad ? "SOME-NOT-JPEG" : "ALL-JPEG-DATA-URLS" }' /tmp/t1.txt
#!must INSIDE-BOUNDS
awk '$2 > 480 || $3 > 300 || $2 < 1 || $3 < 1 { bad = 1 } END { print bad ? "OUT-OF-BOUNDS" : "INSIDE-BOUNDS" }' /tmp/t1.txt
# Aspect: the thumbnail's w/h against ListWindows' w/h (frame, header included), within 3%.
wins > /tmp/w1.txt
#!must ASPECT-MATCHES
awk 'NR == FNR { tw[$1] = $2; th[$1] = $3; next } ($1 in tw) && tw[$1] > 0 { a = $2 / $3; b = tw[$1] / th[$1]; d = a / b - 1; if (d < 0) d = -d; if (d > 0.03) { bad = bad " " $1 ":" a ":" b } n++ } END { print (bad == "" && n >= 5) ? "ASPECT-MATCHES" : "ASPECT-OFF" bad " n=" n }' /tmp/w1.txt /tmp/t1.txt
#!must LAT-COLD-OK
test $((T1 - T0)) -lt 4000 && echo LAT-COLD-OK || echo "LAT-COLD-SLOW $((T1 - T0))"

echo "=== cached: nothing has changed, so nothing is redrawn ==="
T0=$(ms); thumbs '{"max_width":480,"max_height":300}' > /tmp/t2.txt; T1=$(ms)
echo "LAT-CACHED five windows: $((T1 - T0)) ms (includes echo-cli)"
#!must LAT-CACHED-OK
test $((T1 - T0)) -lt 2500 && echo LAT-CACHED-OK || echo "LAT-CACHED-SLOW $((T1 - T0))"
#!must SAME-WINDOWS-AGAIN
test "$(awk '{print $1}' /tmp/t1.txt | sort | tr '\n' ' ')" = "$(awk '{print $1}' /tmp/t2.txt | sort | tr '\n' ' ')" && echo SAME-WINDOWS-AGAIN || echo WINDOWS-CHANGED

echo "=== asked for some ids, gets those ==="
FID=$(wid files); NID=$(wid notes)
thumbs "{\"window_ids\":[$FID,$NID,99999],\"max_width\":200,\"max_height\":200}" | tee /tmp/t3.txt
#!must ONLY-THE-ASKED
test "$(wc -l < /tmp/t3.txt)" = 2 && awk '{ print $1 }' /tmp/t3.txt | grep -qx "$FID" && awk '{ print $1 }' /tmp/t3.txt | grep -qx "$NID" && echo ONLY-THE-ASKED || echo "WRONG-SET $(cat /tmp/t3.txt | wc -l)"
#!must SMALLER-HONOURED
awk '$2 > 200 || $3 > 200 { bad = 1 } END { print bad ? "SMALLER-IGNORED" : "SMALLER-HONOURED" }' /tmp/t3.txt
#!must HUGE-ASK-CLAMPED
thumbs '{"window_ids":['"$FID"'],"max_width":99999,"max_height":99999}' | awk '$2 > 480 || $3 > 480 { bad = 1 } END { print bad ? "NOT-CLAMPED" : "HUGE-ASK-CLAMPED" }'

echo "=== bad arguments are refused ==="
#!must BAD-IDS-REFUSED
echo-cli call org.geckoOS.evo GetWindowThumbnails --interface org.geckoOS.evo --args '{"window_ids":"x"}' 2>&1 | grep -qi 'window_ids must' && echo BAD-IDS-REFUSED || echo BAD-IDS-ACCEPTED

echo "=== a minimized window keeps its picture ==="
echo-cli call org.geckoOS.evo MinimizeWindow --interface org.geckoOS.evo --args "{\"window_id\":$NID}" | grep -c '"success": true'
sleep 3
#!must NOTES-MINIMIZED
echo-cli call org.geckoOS.evo ListWindows --interface org.geckoOS.evo --args '{}' | awk -v id="$NID" '/"visible":/ { gsub(/[",:]/, ""); v = $2 } /"window_id":/ { gsub(/[",:]/, ""); if ($2 == id) print "vis=" v }' | grep -q 'vis=false' && echo NOTES-MINIMIZED || echo NOTES-STILL-VISIBLE
thumbs "{\"window_ids\":[$NID],\"max_width\":480,\"max_height\":300}" | tee /tmp/t4.txt
#!must MINIMIZED-HAS-PICTURE
test "$(wc -l < /tmp/t4.txt)" = 1 && awk '$4 > 500 && $5 == 1 { ok = 1 } END { exit !ok }' /tmp/t4.txt && echo MINIMIZED-HAS-PICTURE || echo MINIMIZED-NO-PICTURE
#!must MINIMIZED-ASPECT-KEPT
awk -v id="$NID" 'NR == FNR { if ($1 == id) { tw = $2; th = $3 } next } { a = $2 / $3; b = tw / th; d = a / b - 1; if (d < 0) d = -d; print (d < 0.03) ? "MINIMIZED-ASPECT-KEPT" : "MINIMIZED-ASPECT-OFF" }' /tmp/w1.txt /tmp/t4.txt
echo-cli call org.geckoOS.evo RestoreWindow --interface org.geckoOS.evo --args "{\"window_id\":$NID}" | grep -c '"success": true'
sleep 2

echo "=== a closed window is gone from the answer ==="
echo-cli call org.geckoOS.nova Close --interface org.geckoOS.nova --args '{"appId":"tasks"}' | grep -E 'success|Error'
sleep 5
#!must CLOSED-GONE
thumbs '{}' | awk '{ print $1 }' | sort > /tmp/t5.txt; wins | awk '{ print $1 }' | sort > /tmp/w5.txt; test "$(comm -23 /tmp/t5.txt /tmp/w5.txt | wc -l)" = 0 && echo CLOSED-GONE || echo "STALE-THUMBNAIL-FOR-GONE-WINDOW"

echo "=== a picture to look at ==="
# The Files window's thumbnail data URL, base64 only, to pull and decode on the host.
echo-cli call org.geckoOS.evo GetWindowThumbnails --interface org.geckoOS.evo --args "{\"window_ids\":[$FID],\"max_width\":480,\"max_height\":300}" | awk '/"data":/ { s = $0; sub(/.*base64,/, "", s); gsub(/[", ]/, "", s); print s }' > /tmp/thumb-files.b64
wc -c /tmp/thumb-files.b64
#!shot thumb-screen
