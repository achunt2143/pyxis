# Tour screenshots: one clean shot of every user-facing feature, for the
# non-technical feature guide. Nothing here is a test: the few `#!must` lines
# only guard a wait that could otherwise fail silently. Each shot is preceded
# by a printed state line, and every shot is named tour-<feature>
# (target/serial-shots/tour-*.png). LOOK at them: a capture succeeding says
# nothing about whether the screen is right.
#
# On an image built after 2026-11-12 nothing needs pushing, except the app
# dists and the dev marker the root dialog needs (the unsigned acceptance
# package is installed through the installer). Full command line, on a dev
# image with everything current:
#
#   : > /tmp/allow-unsigned-packages   # on the host
#   scripts/serial-test.py --memory 4096 \
#     --push target/release/evo:/usr/bin/evo \
#     --push target/release/echo-daemon:/usr/bin/echo-daemon \
#     --push target/release/echo-cli:/usr/bin/echo-cli \
#     --push gecko/patches/pyxis/toolkit/actors/PyxisBridgeParent.sys.mjs:/usr/lib/pyxis-gecko/actors/PyxisBridgeParent.sys.mjs \
#     --push ../pyxis-experience/dist:/usr/share/geckoOS/apps/bloom/webroot \
#     --push ../pyxis-experience/packages/apps/files/dist:/usr/share/geckoOS/apps/files/webroot \
#     --push ../pyxis-experience/packages/apps/notes/dist:/usr/share/geckoOS/apps/notes/webroot \
#     --push ../pyxis-experience/packages/apps/contacts/dist:/usr/share/geckoOS/apps/contacts/webroot \
#     --push ../pyxis-experience/packages/apps/tasks/dist:/usr/share/geckoOS/apps/tasks/webroot \
#     --push ../pyxis-experience/packages/apps/settings/dist:/usr/share/geckoOS/apps/settings/webroot \
#     --push ../pyxis-experience/packages/apps/browser/dist:/usr/share/geckoOS/apps/browser/webroot \
#     --push ../pyxis-experience/packages/apps/terminal/dist:/usr/share/geckoOS/apps/terminal/webroot \
#     --push ../pyxis-experience/packages/apps/installer/dist:/usr/share/geckoOS/apps/installer/webroot \
#     --push /tmp/allow-unsigned-packages:/etc/pyxis/allow-unsigned-packages \
#     --script scripts/serial-tests/tour-screenshots.txt
#
# (An image rebuilt after all of that lands needs none of the --push lines but
# the marker. Bloom's title-bar names and the lock page need the vela whose
# bridge has SetAppIdentity and session.*; on an older vela the bars show the
# client's own title and the lock page is drawn by the stand-in described at
# the shots.)
#
# What is not here, and why:
#   * Nothing skipped. Two things are approximations to look at:
#     - the documents of Notes, Contacts and Tasks are seeded through the
#       filesystem service (scope app:<id>), so each app has real content.
#     - tour-power-dialog reaches Shut Down by Shift+Tab (the panel traps Tab,
#       so Shift+Tab from the first control lands on the last, Shut Down) and
#       Enter, since the button's position moves with the panel's height.
#   * The geckouser password is set for the lock and root-dialog shots and is
#     left set at the end (the lock is then not press-to-enter on this boot).
#
# Run time is long (about an hour): idle lock is turned off at the start.

evo() { local a="${2-}"; [ -n "$a" ] || a='{}'; echo-cli call org.geckoOS.evo "$1" --interface org.geckoOS.evo --args "$a"; }
ostate() { evo GetOverlayState | tr -d ' \n'; }
orects() { ostate | grep -o '"rects":\[[^]]*\]' | grep -o '{[^{}]*}' | while read -r r; do f() { echo "$r" | grep -o "\"$1\":-\?[0-9]*" | cut -d: -f2; }; echo "$(f x) $(f y) $(f width) $(f height)"; done; }
geom() { evo ListWindows | awk -v app="$1" '/"app_id"/ { on = index($0, "\"" app "\"") > 0 } on && /"(x|y|width|height|window_id|focused)":/ { gsub(/[",:]/, ""); v[$1] = $2 } /}/ { if (on) { print v["x"], v["y"], v["width"], v["height"], v["window_id"], v["focused"]; exit } on = 0 }'; }
wid() { evo ListWindows | awk -v app="$1" '/"app_id"/ { on = index($0, "\"" app "\"") > 0 } on && /"window_id":/ { gsub(/[",:]/, ""); print $2; exit }'; }
winact() { echo-cli call org.geckoOS.evo "$1" --interface org.geckoOS.evo --args "{\"window_id\":$2}" | grep -c '"success": true'; }
sset() { echo-cli call org.geckoOS.settings set --interface org.geckoOS.settings --args "{\"key\":\"$1\",\"value\":$2}" | grep -c '"success": true'; }
fscall() { echo-cli call org.geckoOS.filesystem "$1" --interface org.geckoOS.filesystem --args "$2" 2>&1; }
setanim() { echo-cli call org.geckoOS.evo SetAnimationSettings --interface org.geckoOS.evo --args "$1" | grep -c '"mode"'; }
act() { evo GetAnimationState | tr -d ' \n' | grep -o '"active":[0-9]*' | head -1 | cut -d: -f2; }
wait_idle() { local i=0; while [ $i -lt $1 ]; do [ "$(act)" = 0 ] && return 0; sleep 1; i=$((i + 1)); done; echo "STILL-ANIMATING"; return 1; }
blstate() { evo GetLumenState | tr -d ' \n'; }
lfr() { blstate | grep -o '"frames":{[^}]*}' | grep -o "\"$1\":[0-9]*" | cut -d: -f2; }
lpr() { blstate | grep -o '"last_progress":{[^}]*}' | grep -o "\"$1\":[0-9.eE+-]*" | cut -d: -f2; }
# wlp EFFECT FRAMES-BEFORE TENTHS THRESHOLD: wait until EFFECT has drawn a frame beyond the count and reached the progress.
wlp() { local i=0; while [ $i -lt $3 ]; do if [ "$(lfr $1)" -gt "$2" ] && awk -v p="$(lpr $1)" -v w="$4" 'BEGIN { exit !(p + 0 >= w) }'; then echo "$1-REACHED $4 at $(lpr $1)"; return 0; fi; sleep 0.1; i=$((i + 1)); done; echo "$1-MISSED $4 at $(lpr $1)"; return 1; }
bgone() { local i=0; while [ $i -lt 60 ]; do pgrep -f "profiles/$1( |$)" > /dev/null || return 0; sleep 0.5; i=$((i + 1)); done; echo "STILL-RUNNING $1"; return 1; }
bbound() { echo-cli call org.geckoOS.appHost ListRunningApps --interface org.geckoOS.appHost --args '{}' | tr -d ' \n' | grep -o "\"appId\":\"$1\"[^}]*" | grep -q '"surfaces":\[[0-9]'; }
bmapped() { local i=0; while [ $i -lt 80 ]; do [ -n "$(wid $1)" ] && bbound $1 && return 0; sleep 0.5; i=$((i + 1)); done; echo "NOT-MAPPED $1"; return 1; }
blaunch() { echo-cli call org.geckoOS.nova Launch --interface org.geckoOS.nova --args "{\"appId\":\"$1\"}" | grep -E 'sessionId|Error'; }
# lstart: a first run (profile state cleared, so it opens maximized); lopen: launch as it is; lretry: launch again only if the first never mapped.
lstart() { rm -f /var/lib/pyxis/profiles/$1/xulstore.json /var/lib/pyxis/profiles/$1/sessionstore.jsonlz4; blaunch $1; bmapped $1 && echo "UP $1"; }
lopen() { blaunch $1; bmapped $1 && echo "UP $1"; }
lretry() { if [ -n "$(wid $1)" ] && bbound $1; then echo "ALREADY-UP $1"; else echo "RETRY $1"; echo-cli call org.geckoOS.nova Close --interface org.geckoOS.nova --args "{\"appId\":\"$1\"}" > /dev/null 2>&1; bgone $1; blaunch $1; bmapped $1 && echo "UP $1"; fi; }
# flt: float a window (unmaximize); cwin: close it by its own X (its floating place is remembered) and wait for it to be gone.
flt() { winact UnmaximizeWindow $(wid $1) > /dev/null; sleep 3; echo "$1 floats at: $(geom $1)"; }
cwin() { winact CloseWindow $(wid $1) > /dev/null; bgone $1; sleep 2; }
# seed APP STORE JSON: write a document into the app's own storage, as the app would have saved it.
seed() { local esc; esc=$(printf '%s' "$3" | sed 's/\\/\\\\/g; s/"/\\"/g'); fscall writeFile "{\"path\":\"~/$2.json\",\"content\":\"$esc\",\"scope\":\"app:$1\"}" | grep -E 'success|rror' | head -2; }
clear_notes() { for NID in $(echo-cli call org.geckoOS.notification list --interface org.geckoOS.notification --args '{}' | grep -o '"id": "[^"]*"' | cut -d'"' -f4); do echo-cli call org.geckoOS.notification close --interface org.geckoOS.notification --args "{\"notificationId\":\"$NID\"}" | grep -c '"success": true'; done; }
# Lock and root-dialog helpers (from alpha5-lock.txt and alpha5-root-dialog.txt).
lst() { evo GetLockState | tr -d ' \n'; }
lf() { lst | sed 's/"frames":{[^}]*}//' | grep -o "\"$1\":[^,}]*" | head -1 | cut -d: -f2 | tr -d '"'; }
lwait() { local i=0; while [ $i -lt $3 ]; do [ "$(lf $1)" = "$2" ] && { echo "LOCK-$1=$2"; return 0; }; sleep 0.1; i=$((i + 1)); done; echo "LOCK-$1-NOT-$2 ($(lst))"; return 1; }
unlock() { echo-cli call org.geckoOS.session unlock --interface org.geckoOS.session --args "{\"password\":\"$1\"}" | tr -d ' \n'; }
priv() { local a="${2-}"; [ -n "$a" ] || a='{}'; echo-cli call org.geckoOS.privilege "$1" --interface org.geckoOS.privilege --args "$a"; }
asgeck() { su -s /bin/sh geckouser -c "XDG_RUNTIME_DIR=/run/user/1000 $1"; }
bg_install() { rm -f "$1"; ( asgeck "echo-cli call org.geckoOS.installer Install --interface org.geckoOS.installer --args '{\"packagePath\":\"$PKG\"}'"; echo "EXIT=$?" ) > "$1" 2>&1 & }
waitfor() { local i=0; while [ $i -lt $3 ]; do grep -qF -- "$2" "$1" 2>/dev/null && return 0; sleep 0.5; i=$((i + 1)); done; return 1; }
rid() { tr -d ' \n' < "$1" | grep -o '"requestId":"[^"]*"' | tail -1 | cut -d'"' -f4; }
start_watch() { rm -f "$1"; echo-cli watch --signal PrivilegeRequested --seconds 40 > "$1" 2>&1 & sleep 1; }
