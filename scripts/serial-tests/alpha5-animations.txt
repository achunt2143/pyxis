# Alpha 5: window animations (evo/src/anim): open, close, the mode switch.
#
#   scripts/serial-test.py --memory 4096 --push target/release/evo:/usr/bin/evo --push target/release/echo-cli:/usr/bin/echo-cli --script scripts/serial-tests/alpha5-animations.txt
#
# Animation is a render-time transform and the harness can't see motion, so the
# assertions are Evo's own counters (evo.GetAnimationState): every animation
# counts as `started` when it begins and `completed` when it ends however it
# ended, so with nothing in flight `started == completed` and `active == 0`.
# The echo-cli calls are root and need no vela rebuild.
#
# LOOK at target/serial-shots/anim-*.png:
#   anim-open-mid    the Files window part-way through its open: smaller than
#                    its final size, translucent (Bloom's desktop shows
#                    through), its title bar intact and scaled with it.
#   anim-close-mid   the same window mid-close, drawn from a snapshot after
#                    the app's buffer is gone, with its title bar, above the
#                    desktop: since Lumen, Evaporate (see evap-* at the end),
#                    a hole burning out from the close button.
#   anim-close-late  the same ghost six seconds on: the hole larger.
#   geo-*            geometry transitions (maximize, unmaximize, the two
#                    interrupting each other, a work-area change, fullscreen),
#                    each caught part-way at scale 60. Look for: the old
#                    frame's picture, header and all, scaled to the moving
#                    rect with the new window fading in over it; header text
#                    scaled, never re-laid-out; no black frame; and the
#                    *-late shots must show the final frame with no trace of
#                    the old size.
#
#   min-*            minimize and restore (lane B; drawn as Lumen's Droplet and
#                    Rise since lane L1, see droplet-* further down). Look for:
#                    min-mid, the Files window rounded off, soft and tinted on
#                    its way to the dock icon at (600,740); min-late, an
#                    icon-sized orb, fading; restore-mid, the window opening
#                    back out of the orb; min-restore-mid, a minimize
#                    turned round mid-flight with no jump in size or place;
#                    min-nodock, a window with no reported icon heading for the
#                    bottom centre of the screen; showdesk-*, three windows, the
#                    front-most already well on its way, the next less, the
#                    last barely started (45 ms x scale apart, 1.35 s at scale
#                    30), drawn as Lumen's Recede (see recede-* there); closemid-*, a window closed while it was still
#                    opening, evaporating from the size it had (never first
#                    jumping to full size).
#
# The mid-animation shots run at scale 60 (a 260 ms open takes 15.6 s, a close
# 10.8 s), slow enough for the guest's shell and the screenshot to land inside.

# "started-open completed-open started-close completed-close active mode"
st() { echo-cli call org.geckoOS.evo GetAnimationState --interface org.geckoOS.evo --args '{}' | awk '/"started"/ { sec = "s" } /"completed"/ { sec = "c" } /"(open|close)":/ { gsub(/[",:]/, ""); v[sec $1] = $2 } /"active"/ { gsub(/[",:]/, ""); a = $2 } /"mode"/ { gsub(/[",:]/, ""); m = $2 } END { print v["sopen"] + 0, v["copen"] + 0, v["sclose"] + 0, v["cclose"] + 0, a + 0, m }'; }
# "window_id" of the first window whose app_id is $1.
wid() { echo-cli call org.geckoOS.evo ListWindows --interface org.geckoOS.evo --args '{}' | awk -v app="$1" '/"app_id"/ { on = index($0, "\"" app "\"") > 0 } on && /"window_id":/ { gsub(/[",:]/, ""); print $2; exit }'; }
# "started completed" of the geometry kind, then the shared active count.
gst() { echo-cli call org.geckoOS.evo GetAnimationState --interface org.geckoOS.evo --args '{}' | awk '/"started"/ { sec = "s" } /"completed"/ { sec = "c" } /"geometry":/ { gsub(/[",:]/, ""); v[sec] = $2 } /"active"/ { gsub(/[",:]/, ""); a = $2 } END { print v["s"] + 0, v["c"] + 0, a + 0 }'; }
# "x y w h id" of the first window whose app_id is $1.
geom() { echo-cli call org.geckoOS.evo ListWindows --interface org.geckoOS.evo --args '{}' | awk -v app="$1" '/"app_id"/ { on = index($0, "\"" app "\"") > 0 } on && /"(x|y|width|height|window_id)":/ { gsub(/[",:]/, ""); v[$1] = $2 } /}/ { if (on) { print v["x"], v["y"], v["width"], v["height"], v["window_id"]; exit } on = 0 }'; }
# The edges Bloom reserved, "top bottom".
wa() { echo-cli call org.geckoOS.evo GetOverlayState --interface org.geckoOS.evo --args '{}' | awk '/"top"/ { gsub(/[",:]/, ""); t = $2 } /"bottom"/ { gsub(/[",:]/, ""); b = $2 } END { print t + 0, b + 0 }'; }
winact() { echo-cli call org.geckoOS.evo "$1" --interface org.geckoOS.evo --args "{\"window_id\":$2}" | grep -c '"success": true'; }
setanim() { echo-cli call org.geckoOS.evo SetAnimationSettings --interface org.geckoOS.evo --args "$1" | grep -c '"mode"'; }
setwa() { echo-cli call org.geckoOS.evo SetWorkArea --interface org.geckoOS.evo --args "{\"top\":$1,\"bottom\":$2}" | grep -c success; }
# Wait until the geometry kind is active (the transition has begun).
wait_geo() { i=0; while [ $i -lt $1 ]; do read _ _ A <<< "$(gst)"; [ "$A" != 0 ] && return 0; sleep 0.5; i=$((i + 1)); done; return 1; }
# Wait (up to $1 half-seconds) until animations are active.
wait_active() { i=0; while [ $i -lt $1 ]; do read _ _ _ _ A _ <<< "$(st)"; [ "$A" != 0 ] && return 0; sleep 0.5; i=$((i + 1)); done; return 1; }
# Wait until nothing is animating.
wait_idle() { i=0; while [ $i -lt $1 ]; do read _ _ _ _ A _ <<< "$(st)"; [ "$A" = 0 ] && return 0; sleep 1; i=$((i + 1)); done; return 1; }

echo "=== settings: the IPC contract ==="
#!must SETTINGS-OK
echo-cli call org.geckoOS.evo SetAnimationSettings --interface org.geckoOS.evo --args '{"mode":"full","scale":1}' | grep -q '"mode": "full"' && echo SETTINGS-OK || echo SETTINGS-FAILED
#!must BAD-MODE-REFUSED
echo-cli call org.geckoOS.evo SetAnimationSettings --interface org.geckoOS.evo --args '{"mode":"wobbly"}' 2>&1 | grep -qi 'mode must be' && echo BAD-MODE-REFUSED || echo BAD-MODE-ACCEPTED
#!must STATE-SHAPE-OK
echo-cli call org.geckoOS.evo GetAnimationState --interface org.geckoOS.evo --args '{}' | grep -E '"(mode|scale|active|started|completed|open|close|minimize|restore|geometry)"' | wc -l | awk '{ print ($1 >= 14) ? "STATE-SHAPE-OK" : "STATE-SHAPE-BAD " $1 }'
read SO0 CO0 SC0 CC0 A0 MODE0 <<< "$(st)"; echo "baseline: open $SO0/$CO0 close $SC0/$CC0 active=$A0 mode=$MODE0"
#!must BASELINE-MODE-FULL
test "$MODE0" = full && echo BASELINE-MODE-FULL || echo "BASELINE-MODE-$MODE0"

echo "=== open: one animation for a launched app, none for the pre-window ==="
rm -f /var/lib/pyxis/profiles/files/xulstore.json /var/lib/pyxis/profiles/files/sessionstore.jsonlz4; echo first-run
#!must sessionId
echo-cli call org.geckoOS.nova Launch --interface org.geckoOS.nova --args '{"appId":"files"}' | grep -E 'sessionId|Error'
sleep 18
wait_idle 10
read SO1 CO1 SC1 CC1 A1 _ <<< "$(st)"; echo "after launch: open $SO1/$CO1 close $SC1/$CC1 active=$A1"
# Exactly one: the app's window. The about:blank pre-window each launch maps
# first claims vela-default, which is not bindable, and must not animate.
#!must OPEN-ONCE
test $((SO1 - SO0)) = 1 && echo OPEN-ONCE || echo "OPEN-COUNT-$((SO1 - SO0))"
#!must OPEN-COMPLETED
test "$SO1" = "$CO1" && echo OPEN-COMPLETED || echo "OPEN-NOT-COMPLETED $SO1/$CO1"
#!must IDLE-AFTER-OPEN
test "$A1" = 0 && echo IDLE-AFTER-OPEN || echo "ACTIVE-$A1"
# Bloom never animates, so a desktop-only boot has started nothing of its own:
# the count above is the app's alone.

echo "=== close: a ghost drawn from a snapshot, then gone ==="
FID=$(wid files); echo "files window $FID"
#!must CLOSE-REQUESTED
echo-cli call org.geckoOS.evo CloseWindow --interface org.geckoOS.evo --args "{\"window_id\":$FID}" | grep -q '"success": true' && echo CLOSE-REQUESTED || echo CLOSE-REFUSED
sleep 3
wait_idle 10
read SO2 CO2 SC2 CC2 A2 _ <<< "$(st)"; echo "after close: open $SO2/$CO2 close $SC2/$CC2 active=$A2"
#!must CLOSE-STARTED
test $((SC2 - SC1)) = 1 && echo CLOSE-STARTED || echo "CLOSE-COUNT-$((SC2 - SC1))"
#!must CLOSE-COMPLETED
test "$SC2" = "$CC2" && echo CLOSE-COMPLETED || echo "CLOSE-NOT-COMPLETED $SC2/$CC2"
#!must IDLE-AFTER-CLOSE
test "$A2" = 0 && echo IDLE-AFTER-CLOSE || echo "ACTIVE-$A2"
# The window is gone from Evo's own list: the ghost took no part in it.
#!must WINDOW-GONE
test -z "$(wid files)" && echo WINDOW-GONE || echo WINDOW-STILL-LISTED
# Make sure the app has really exited before the next launch.
echo-cli call org.geckoOS.nova Close --interface org.geckoOS.nova --args '{"appId":"files"}' | grep -E 'success|Error'
sleep 3

echo "=== off: the counters do not move ==="
#!must OFF-OK
echo-cli call org.geckoOS.evo SetAnimationSettings --interface org.geckoOS.evo --args '{"mode":"off"}' | grep -q '"mode": "off"' && echo OFF-OK || echo OFF-FAILED
read SO3 CO3 SC3 CC3 A3 MODE3 <<< "$(st)"; echo "off: open $SO3/$CO3 close $SC3/$CC3 mode=$MODE3"
echo-cli call org.geckoOS.nova Launch --interface org.geckoOS.nova --args '{"appId":"files"}' | grep -E 'sessionId|Error'
sleep 18
FID=$(wid files); echo "files window $FID"
#!must FILES-MAPPED-OFF
test -n "$FID" && echo FILES-MAPPED-OFF || echo FILES-NOT-MAPPED
echo-cli call org.geckoOS.evo CloseWindow --interface org.geckoOS.evo --args "{\"window_id\":$FID}" | grep -q '"success": true'
sleep 4
read SO4 CO4 SC4 CC4 A4 _ <<< "$(st)"; echo "after an off launch and close: open $SO4/$CO4 close $SC4/$CC4 active=$A4"
#!must OFF-COUNTERS-STILL
test "$SO4 $CO4 $SC4 $CC4 $A4" = "$SO3 $CO3 $SC3 $CC3 $A3" && echo OFF-COUNTERS-STILL || echo "OFF-COUNTERS-MOVED"
echo-cli call org.geckoOS.nova Close --interface org.geckoOS.nova --args '{"appId":"files"}' | grep -E 'success|Error'
sleep 3

echo "=== reduced: opacity only, still counted ==="
#!must REDUCED-OK
echo-cli call org.geckoOS.evo SetAnimationSettings --interface org.geckoOS.evo --args '{"mode":"reduced","scale":1}' | grep -q '"mode": "reduced"' && echo REDUCED-OK || echo REDUCED-FAILED
echo-cli call org.geckoOS.nova Launch --interface org.geckoOS.nova --args '{"appId":"files"}' | grep -E 'sessionId|Error'
sleep 18
wait_idle 10
read SO5 CO5 _ _ A5 _ <<< "$(st)"; echo "reduced launch: open $SO5/$CO5 active=$A5"
#!must REDUCED-OPEN-COUNTED
test $((SO5 - SO4)) = 1 -a "$SO5" = "$CO5" -a "$A5" = 0 && echo REDUCED-OPEN-COUNTED || echo "REDUCED-OPEN-WRONG"
echo-cli call org.geckoOS.nova Close --interface org.geckoOS.nova --args '{"appId":"files"}' | grep -E 'success|Error'
sleep 3

echo "=== slow motion: look at a window part-way through opening and closing ==="
echo-cli call org.geckoOS.evo SetAnimationSettings --interface org.geckoOS.evo --args '{"mode":"full","scale":60}' | grep -c '"scale": 60'
echo-cli call org.geckoOS.nova Launch --interface org.geckoOS.nova --args '{"appId":"files"}' | grep -E 'sessionId|Error'
# The window maps about 15 s after the launch; the open then runs 15.6 s. The
# shot goes right after the first sight of it: expo-out is most of the way
# there within the first few percent of its time.
#!must OPEN-IN-FLIGHT
wait_active 80 && echo OPEN-IN-FLIGHT || echo OPEN-NEVER-STARTED
#!shot anim-open-mid
read _ _ _ _ AM _ <<< "$(st)"; echo "mid-open active=$AM"
# The yardstick for the geometry fps below: the same VM drawing an open.
sleep 2
echo "fps during an open:"; echo-cli call org.geckoOS.evo GetRenderingStatus --interface org.geckoOS.evo --args '{}' | grep -E '"fps"'
wait_idle 40
FID=$(wid files); echo "files window $FID"
sleep 1
echo-cli call org.geckoOS.evo CloseWindow --interface org.geckoOS.evo --args "{\"window_id\":$FID}" | grep -q '"success": true'
#!must CLOSE-IN-FLIGHT
wait_active 20 && echo CLOSE-IN-FLIGHT || echo CLOSE-NEVER-STARTED
#!shot anim-close-mid
sleep 6
#!shot anim-close-late
wait_idle 40
echo-cli call org.geckoOS.nova Close --interface org.geckoOS.nova --args '{"appId":"files"}' | grep -E 'success|Error'

echo "=== geometry: maximize, unmaximize, a work-area change ==="
# Bloom reports the work area through the bridge; on an older vela say it for
# Bloom (does nothing once it has).
read WT0 WB0 <<< "$(wa)"; [ "$WT0" != 0 ] || setwa 44 76
read WT WB <<< "$(wa)"; MAXH=$((800 - WT - WB)); echo "work area: top=$WT bottom=$WB maximized height=$MAXH"
setanim '{"mode":"full","scale":1}'
rm -f /var/lib/pyxis/profiles/files/xulstore.json /var/lib/pyxis/profiles/files/sessionstore.jsonlz4; echo first-run
echo-cli call org.geckoOS.nova Launch --interface org.geckoOS.nova --args '{"appId":"files"}' | grep -E 'sessionId|Error'
sleep 18
wait_idle 10
FID=$(wid files); echo "files window $FID"
# A first-run window may be born maximized: leave it floating to start from.
winact UnmaximizeWindow $FID | grep -q 1
sleep 3
wait_idle 10
read FX FY FW FH FID2 <<< "$(geom files)"; echo "floating: $FX,$FY ${FW}x$FH"

# --- maximize: exactly one transition, and it lands on the work area.
read GS0 GC0 GA0 <<< "$(gst)"
#!must MAX-REQUESTED
winact MaximizeWindow $FID | grep -q 1 && echo MAX-REQUESTED || echo MAX-REFUSED
sleep 3
wait_idle 10
read GS1 GC1 GA1 <<< "$(gst)"; echo "maximize: geometry $GS0/$GC0 -> $GS1/$GC1 active=$GA1"
#!must GEO-MAX-ONE
test $((GS1 - GS0)) = 1 && echo GEO-MAX-ONE || echo "GEO-MAX-COUNT-$((GS1 - GS0))"
#!must GEO-MAX-BALANCED
test "$GS1" = "$GC1" -a "$GA1" = 0 && echo GEO-MAX-BALANCED || echo "GEO-MAX-UNBALANCED $GS1/$GC1 active=$GA1"
read AX AY AW AH _ <<< "$(geom files)"; echo "maximized: $AX,$AY ${AW}x$AH"
#!must GEO-MAX-FRAME
test "$AX,$AY ${AW}x$AH" = "0,$WT 1280x$MAXH" && echo GEO-MAX-FRAME || echo "GEO-WRONG-FRAME $AX,$AY ${AW}x$AH (want 0,$WT 1280x$MAXH)"
#!shot geo-maximized

# --- unmaximize: one more, and back to the floating frame it left.
#!must UNMAX-REQUESTED
winact UnmaximizeWindow $FID | grep -q 1 && echo UNMAX-REQUESTED || echo UNMAX-REFUSED
sleep 3
wait_idle 10
read GS2 GC2 GA2 <<< "$(gst)"; echo "unmaximize: geometry $GS1/$GC1 -> $GS2/$GC2 active=$GA2"
#!must GEO-UNMAX-ONE
test $((GS2 - GS1)) = 1 -a "$GS2" = "$GC2" -a "$GA2" = 0 && echo GEO-UNMAX-ONE || echo "GEO-UNMAX-WRONG $GS2/$GC2 active=$GA2"
read UX UY UW UH _ <<< "$(geom files)"; echo "unmaximized: $UX,$UY ${UW}x$UH"
#!must GEO-UNMAX-FRAME
test "$UX,$UY ${UW}x$UH" = "$FX,$FY ${FW}x$FH" && echo GEO-UNMAX-FRAME || echo "GEO-NOT-RESTORED $UX,$UY ${UW}x$UH (was $FX,$FY ${FW}x$FH)"

# --- interactive move stays 1:1: it starts no transition.
SX=$((UX + UW / 2)); SY=$((UY + 12)); EX=$((SX - 30)); EY=$SY
#!pointer drag $SX $SY $EX $EY
sleep 2
read DX DY _ _ _ <<< "$(geom files)"
read GS3 GC3 GA3 <<< "$(gst)"
#!must DRAG-NO-TRANSITION
test "$DX" = $((UX - 30)) -a "$GS3" = "$GS2" && echo DRAG-NO-TRANSITION || echo "DRAG-ANIMATED-OR-FAILED x=$DX started=$GS3 (was $GS2)"
winact MaximizeWindow $FID | grep -q 1
sleep 3
wait_idle 10
winact UnmaximizeWindow $FID | grep -q 1
sleep 3
wait_idle 10
read FX FY FW FH _ <<< "$(geom files)"; echo "floating again: $FX,$FY ${FW}x$FH"

# --- a work-area change refits the maximized window with a transition.
winact MaximizeWindow $FID | grep -q 1
sleep 3
wait_idle 10
read GS4 GC4 _ <<< "$(gst)"
NT=$((WT + 20))
setwa $NT $WB | grep -q 1
sleep 3
wait_idle 10
read GS5 GC5 GA5 <<< "$(gst)"
read RX RY RW RH _ <<< "$(geom files)"; echo "after the bar grew: $RX,$RY ${RW}x$RH geometry $GS4 -> $GS5"
#!must GEO-REFIT
test $((GS5 - GS4)) -ge 1 -a "$GS5" = "$GC5" -a "$GA5" = 0 && echo GEO-REFIT || echo "GEO-REFIT-WRONG $GS4 -> $GS5 completed=$GC5 active=$GA5"
#!must GEO-REFIT-FRAME
test "$RX,$RY ${RW}x$RH" = "0,$NT 1280x$((800 - NT - WB))" && echo GEO-REFIT-FRAME || echo "GEO-REFIT-WRONG-FRAME $RX,$RY ${RW}x$RH"
setwa $WT $WB | grep -q 1
sleep 3
wait_idle 10
read RX RY RW RH _ <<< "$(geom files)"
#!must GEO-REFIT-BACK
test "$RX,$RY ${RW}x$RH" = "0,$WT 1280x$MAXH" && echo GEO-REFIT-BACK || echo "GEO-REFIT-NOT-BACK $RX,$RY ${RW}x$RH"
winact UnmaximizeWindow $FID | grep -q 1
sleep 3
wait_idle 10

#!must GEO-OFF-FRAME

setanim '{"mode":"full","scale":60}' | grep -q 1

#!must GEO-MAX-IN-FLIGHT

# The client takes a moment to paint its new size; then the spring runs for

sleep 3

read _ _ AM <<< "$(gst)"; echo "mid-maximize active=$AM"

wait_idle 60
read GMS GMC GMA <<< "$(gst)"
#!must GEO-INTERRUPT-BALANCED

sleep 4

echo "=== Lumen Settle glint: a band of light crosses the frame as a geometry change ends ==="
# Lane L4 (evo/src/anim/glint.rs). From 42% of the transition to its end a
# soft diagonal white band (peak alpha 0.32 x sin(pi g)) crosses the window,
# clipped to its frame. Full motion only, and only with Lumen on. Asserted
# from GetLumenState (`frames.glint`, `last_progress.glint`, `glint_program`).
#
# LOOK at target/serial-shots/glint-*.png (maximize at scale 60, the band
# caught at 25/50/75% of its crossing):
#   glint-25   a faint pale band entering from the top-left corner of the
#              nearly-maximized window, nothing outside the frame.
#   glint-50   the band across the middle, at its brightest: the window still
#              readable through it, no hard edge.
#   glint-75   the band leaving by the bottom-right, fading.

echo "=== Lumen Droplet and Rise: minimize into an orb, restore out of it (evo/src/anim/droplet.rs, warp.rs, lumen.rs) ==="
# Lane L1. A minimize draws the window's snapshot through the Lumen program:
# it rounds off into a circle the size of its dock icon over the first 60%,
# goes soft (blur up to 6 px), takes on the app's accent (tint and an outer
# glow by ~50%), travels an arc that lifts ~70 px above the higher end,
# squashing and stretching along its velocity once it is orb-like, and lands:
# the last 18% shrinks it by 55%, the last 14% fades it. At 90% Evo sends
# `MinimizeLanded {phase: "land"}`; a restore (Rise, the same backwards) sends
# `rise` as it starts. The same minimize spring drives it, so a restore that
# interrupts a minimize cannot jump. The assertions are GetLumenState
# (`frames.droplet`, `program`, `last_progress.droplet`, `landed`), the
# animation counters, and the signal itself, read with `echo-cli watch`
# (a new echo-cli: push target/release/echo-cli:/usr/bin/echo-cli with this
# suite). Shots are taken when the Droplet's own progress reaches 25/50/75%, at
# scale 13 (the 720 ms spring takes 9.4 s; a spring's run is capped at 10 s).
# The Files window's accent is sent as #2e9e6b (green) so the orb's colour is
# Bloom's, not the window's average. LOOK at target/serial-shots/droplet-*.png:
#   droplet-below-25    the window already smaller and clearly rounded at every
#                       corner, a little soft, faintly green, a few px of glow
#                       round it; heading down along a slight arc.
#   droplet-below-50    nearly a circle, green, blurred, glowing, a little
#                       stretched along the way it moves (down toward the icon).
#   droplet-below-75    a green orb about icon-sized over or just above the
#                       icon at (600,740), glowing; no window left anywhere.
#   droplet-left-* / droplet-right-*  the same toward icons left and right:
#                       the path lifts and arcs over toward the icon, the orb
#                       stretched along the arc, never a funnel or a fold.
#   rise-75/50/25       a restore: the orb leaves the icon and opens back into
#                       the window, sharp and untinted by 25%.
#   droplet-interrupt-* a restore that turned a minimize round at 50% and a
#                       minimize that turned a restore round: shape, place and
#                       colour carry on from where they were, no jump.
#   droplet-zorder      two windows in flight: the one in front stays in front.
#   droplet-lite-50     Lumen off: genie-lite's plain scaled window, no orb.
lstate_d() { echo-cli call org.geckoOS.evo GetLumenState --interface org.geckoOS.evo --args '{}'; }
# A Droplet field: $1 = frames|last_progress|landed, $2 = key.
lfield_d() { lstate_d | awk -v want="$1" -v key="$2" '/"frames"/ { s = "frames" } /"last_progress"/ { s = "last_progress" } /"landed"/ { s = "landed" } $0 ~ "\"" key "\":" { gsub(/[",:]/, ""); if (s == want) { print $2 + 0; exit } }'; }
dframes() { lfield_d frames droplet; }
dprog() { lfield_d last_progress droplet; }
dland() { lfield_d landed land; }
drise() { lfield_d landed rise; }
dprogram() { lstate_d | awk '/"program":/ { gsub(/[",:]/, ""); print $2; exit }'; }
dl_setlumen() { echo-cli call org.geckoOS.evo SetLumen --interface org.geckoOS.evo --args "{\"enabled\":$1}" | grep -c '"enabled"'; }
# Wait (up to $3 polls) for the Droplet's progress to reach $1 going $2 (up|down).
wait_dp() { i=0; while [ $i -lt $3 ]; do P=$(dprog); if [ "$2" = up ]; then awk -v p="$P" -v w="$1" 'BEGIN { exit !(p >= w) }' && return 0; else awk -v p="$P" -v w="$1" 'BEGIN { exit !(p <= w) }' && return 0; fi; i=$((i + 1)); done; return 1; }
# Wait (up to $2 polls) until more frames than $1 went through the Droplet.
wait_dframes() { i=0; while [ $i -lt $2 ]; do [ "$(dframes)" -gt "$1" ] && return 0; i=$((i + 1)); done; return 1; }
wait_midle_d() { i=0; while [ $i -lt $1 ]; do read _ _ _ _ A <<< "$(mst)"; [ "$A" = 0 ] && return 0; sleep 1; i=$((i + 1)); done; return 1; }
settargets_d() { echo-cli call org.geckoOS.evo SetMinimizeTargets --interface org.geckoOS.evo --args "$1" | grep -c '"ok": true'; }

eprog() { lstate | awk '/"frames"/ { s = "f" } /"last_progress"/ { s = "p" } /"evaporate":/ { gsub(/[",:]/, ""); if (s == "p") print $2 + 0 }'; }

# Wait (up to $2 tenths of a second) until more frames than $1 went through

# runs out its count in a second or two and the shots land early.
wait_eframes() { i=0; while [ $i -lt $2 ]; do [ "$(eframes)" -gt "$1" ] && return 0; sleep 0.1; i=$((i + 1)); done; return 1; }
# Wait (up to $2 tenths of a second) until Evaporate's progress reaches $1;
# says EP-REACHED or EP-MISSED so a suite can assert the shot is where it says.
wait_ep() { i=0; while [ $i -lt $2 ]; do awk -v p="$(eprog)" -v w="$1" 'BEGIN { exit !(p >= w) }' && { echo "EP-REACHED $1 at $(eprog)"; return 0; }; sleep 0.1; i=$((i + 1)); done; echo "EP-MISSED $1 at $(eprog)"; return 1; }

evap_close() { echo-cli call org.geckoOS.evo CloseWindow --interface org.geckoOS.evo --args "{\"window_id\":$(wid $1)}" | grep -q '"success": true'; }

# Lane L3. An open is drawn as Bloom: the live window, blurred 10 px -> 0 (the
# focus pull), tinted toward its accent, its corners 42 -> 16, over a radial
# glow of the accent that peaks half-way, while it grows from 0.90 on a spring
# and rises 16 px. The asserts are GetLumenState's `frames.bloom` (and
# `bloom_blurred_frames`, `bloom_program`) and the open counters; shots are
# taken when `last_progress.bloom` reaches 8/25/50/75% at scale 60 (a 25 s
# open). LOOK at target/serial-shots/bloom-*.png:
#   bloom-08   just mapped: very soft, washed toward cool white, the corners
#              much rounder than 16 px, the silhouette itself soft, a faint

blstate() { echo-cli call org.geckoOS.evo GetLumenState --interface org.geckoOS.evo --args '{}' | tr -d ' \n'; }
# The "bloom" entry of the object named $1 (frames, last_progress).
lfield() { blstate | grep -o "\"$1\":{[^}]*}" | grep -o '"bloom":[0-9.eE+-]*' | cut -d: -f2; }
bframes() { F=$(lfield frames); echo "${F:-0}"; }
bprog() { P=$(lfield last_progress); echo "${P:-0}"; }
bblurred() { B=$(blstate | grep -o '"bloom_blurred_frames":[0-9]*' | cut -d: -f2); echo "${B:-0}"; }
bprogram() { blstate | grep -o '"bloom_program":"[a-z]*"' | cut -d'"' -f4; }
setbloom() { echo-cli call org.geckoOS.evo SetLumen --interface org.geckoOS.evo --args "$1" > /dev/null; }
# Wait (up to $2 polls) until more Bloom frames than $1 have been drawn.
wait_bf() { i=0; while [ $i -lt $2 ]; do [ "$(bframes)" -gt "$1" ] && return 0; sleep 0.2; i=$((i + 1)); done; return 1; }
# Wait (up to $2 polls, 20 a second) until Bloom's progress reaches $1.
wait_bp() { i=0; while [ $i -lt $2 ]; do awk -v p="$(bprog)" -v w="$1" 'BEGIN { exit !(p >= w) }' && return 0; sleep 0.05; i=$((i + 1)); done; return 1; }
# Wait (up to 30 s) until no process of app $1 is left (vela and its pyxis-wlctx),
# so each launch starts from nothing rather than racing the last one's exit.
bgone() { i=0; while [ $i -lt 60 ]; do pgrep -f "profiles/$1( |$)" > /dev/null || return 0; sleep 0.5; i=$((i + 1)); done; echo "STILL-RUNNING $1"; return 1; }
bclose() { echo-cli call org.geckoOS.nova Close --interface org.geckoOS.nova --args "{\"appId\":\"$1\"}" | grep -E 'success|Error'; bgone $1; }
# Wait (up to 40 s) until app $1 has its real window: one appHost has bound.
# ListWindows lists vela's about:blank pre-window under the app's id too, and
# a wait that ended on it let the checks read the counters before the open.
bbound() { echo-cli call org.geckoOS.appHost ListRunningApps --interface org.geckoOS.appHost --args '{}' | tr -d ' \n' | grep -o "\"appId\":\"$1\"[^}]*" | grep -q '"surfaces":\[[0-9]'; }

# Launch app $1 and wait for its window, saying so with the guest's log and
# trying once more if none comes. The vela that "never mapped" in about half the

# counts as a start-up crash, and past three a chromeless vela exited at once.

blaunch() { echo-cli call org.geckoOS.nova Launch --interface org.geckoOS.nova --args "{\"appId\":\"$1\"}" | grep -E 'sessionId|Error'; }
