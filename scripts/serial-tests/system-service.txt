# org.geckoOS.system — the service behind the bridge's getSystemInfo()/
# listServices(), added 2026-09-20.
#
#   scripts/serial-test.py --script scripts/serial-tests/system-service.txt
#
# The point of the service is that an app can call those two methods and get
# an answer, so the check that matters is not "echo-cli prints something" —
# it is the third-party app's own probe (see hello-pyxis/README.md), which
# recorded `getSystemInfo=FAILED ... Service not registered` before this
# existed.

# ---------- the service is on the bus at all ----------
#!must "pong": true
echo-cli call org.geckoOS.system ping --interface org.geckoOS.system --args '{}'
#!must "uptime"
#!must "kernel"
#!must "load"
#!must-not Error
echo-cli call org.geckoOS.system getInfo --interface org.geckoOS.system --args '{}'

# `unavailable` is present only when a sub-fact failed, so an empty grep here
# is the good outcome — and any hit names what could not be read.
#!must-not unavailable
echo-cli call org.geckoOS.system getInfo --interface org.geckoOS.system --args '{}' | grep -A6 unavailable

# ---------- listServices sees in-process services, which the old one did not ----------
# org.geckoOS.Echo's ListServices reads ServiceRegistry alone and answers
# "(no services registered)" on a booted device. Both are run here so the
# difference is on the record rather than asserted.
echo-cli call org.geckoOS.Echo ListServices --interface Bus --args '{}' | head -12
# The whole reason this service exists: the old handler sees only what
# registered over the wire, and answers in single digits; this one sees the
# router too. Phrased as a word because assertions are substring matches,
# not regexes — `#!must-not ^[0-4]$` would pass vacuously.
#
# The sentinel must stay UNDER 8 CHARACTERS. It used to be
# SEES-IN-PROCESS-SERVICES, which the command-echo stripper ate -- the sentinel
# a command prints is by definition a substring of the command that prints it,
# and the stripper drops head lines of 8+ characters that match. This assertion
# therefore failed on every passing run. See serial-test.py's docstring.
#!must MANY
test $(echo-cli call org.geckoOS.system listServices --interface org.geckoOS.system --args '{}' | grep -c '"destination"') -gt 8 && echo MANY || echo FEW
#!must org.geckoOS.nova
#!must org.geckoOS.filesystem
#!must org.geckoOS.installer
#!must org.geckoOS.system
#!must org.geckoOS.evo
echo-cli call org.geckoOS.system listServices --interface org.geckoOS.system --args '{}' | grep -o '"destination": "[^"]*"' | sort
# Both kinds must be represented: builtin (router handlers) and external
# (socket clients like Evo). Only one would mean a source was missed.
#!must builtin
#!must external
echo-cli call org.geckoOS.system listServices --interface org.geckoOS.system --args '{}' | grep -o '"kind": "[^"]*"' | sort | uniq -c
echo-cli call org.geckoOS.system listServices --interface org.geckoOS.system --args '{}' | grep -c '"reachable": true'

# ---------- the repointed CLI subcommands ----------
#!must Total:
#!must reachable
echo-cli services 2>&1 | tail -12
echo-cli status 2>&1 | tail -8
#!must Pong
echo-cli ping 2>&1 | tail -4

# ---------- the actual acceptance test: a third-party app calling it ----------
# hello-pyxis probes the bridge on load and writes the result to a file.
# Before this service existed the probe recorded
#   getSystemInfo=FAILED Error: Method failed: Service not registered: org.geckoOS.system
rm -f /var/lib/pyxis/data/com.pyxis.hello/proof-*.txt
echo-cli call org.geckoOS.installer Install --interface org.geckoOS.installer --args '{"packagePath":"/var/lib/pyxis/incoming/com.pyxis.hello_0.2.0_all.ipk"}' | grep -E 'registeredApps|launchable'
#!must sessionId
echo-cli call org.geckoOS.nova Launch --interface org.geckoOS.nova --args '{"appId":"com.pyxis.hello"}' | grep sessionId
sleep 15
# The page writes its proof once every probe has answered, which a
# freshly built image on a loaded host can take longer than 15 s to do:
# wait up to 45 s more for it.
for i in $(seq 1 45); do ls /var/lib/pyxis/data/com.pyxis.hello/proof-*.txt > /dev/null 2>&1 && break; sleep 1; done; echo "proof waited ${i}s more"
# The acceptance check: a THIRD-PARTY app calling the new service through
# the bridge. This line read "getSystemInfo=FAILED ... Service not
# registered: org.geckoOS.system" before the service existed.
# The page writes ONE FILE PER LAUNCH, in its own storage since filesystem
# calls were confined (2026-10-06): /var/lib/pyxis/data/com.pyxis.hello/proof-<ms>.txt,
# not a fixed name (see hello-pyxis/public/index.html: appendFile wrote nothing at
# all on device, so each launch records itself instead). This read globbed the
# old fixed name until 2026-09-25 and so asserted on a file that is never
# created: `cat` printed "No such file or directory" and all three assertions
# below failed on a perfectly good run, for weeks. Cat the glob, and assert the
# file count too so a missing probe cannot pass quietly.
#!must getSystemInfo=ok
#!must bridge=present
cat /var/lib/pyxis/data/com.pyxis.hello/proof-*.txt 2>&1
#!must 1
ls /var/lib/pyxis/data/com.pyxis.hello/proof-*.txt 2>/dev/null | wc -l

# ---------- clean up so the image is left as it was ----------
echo-cli call org.geckoOS.nova Close --interface org.geckoOS.nova --args '{"appId":"com.pyxis.hello"}' | grep appId
echo-cli call org.geckoOS.installer Remove --interface org.geckoOS.installer --args '{"package":"com.pyxis.hello"}' | grep registeredApps
