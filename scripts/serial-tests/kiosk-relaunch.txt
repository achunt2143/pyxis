# Kiosk lost on relaunch after a graceful window close.
#
#   scripts/serial-test.py --script scripts/serial-tests/kiosk-relaunch.txt
#
# Reported symptom: launch an app (chromeless, correct), close it with the
# titlebar X, launch it again -> full browser chrome.
#
# The close method is the whole point. `nova.Close` signals the process
# through AppHost and relaunches cleanly — verified, not the bug. The
# titlebar X sends `xdg_toplevel.close`, letting Gecko run its own
# teardown, and that is the path that leaves state behind. Driving it needs
# `evo.CloseWindow`, which was a stub until 2026-09-21 and never sent the
# client anything, which is why this went unreproduced for so long.
#
# Full browser chrome means `openPyxisAppWindow` did not run and the normal
# `openBrowserWindow` path did. The PYXIS-DIAG probes compiled into vela
# log `cmdLine.state` and the app id at handle() time, which distinguishes
# the candidates:
#   state=0 (INITIAL_LAUNCH) + app id present  -> should have gone kiosk;
#                                                 look inside the handler
#   state=2 (REMOTE_AUTO)                      -> remoted into a surviving
#                                                 vela; MOZ_NO_REMOTE is
#                                                 not doing its job
#   app id null                                -> the env/flag was lost on
#                                                 the second exec

echo "=== LAUNCH 1 ==="
#!must sessionId
echo-cli call org.geckoOS.nova Launch --interface org.geckoOS.nova --args '{"appId":"files"}' | grep -E 'sessionId|Error'
sleep 18
#!must windowclass=files
grep -a 'PYXIS-DIAG' /var/log/echo.log | tail -6
WID=$(echo-cli call org.geckoOS.nova Focus --interface org.geckoOS.nova --args '{"appId":"files"}' | grep -o '"windowId": [0-9]*' | grep -o '[0-9]*'); echo "WID=$WID"
VELA1=$(pgrep -f '[-]-pyxis-app files' | head -1); echo "vela1=$VELA1"

echo "=== CLOSE VIA THE WINDOW, as the titlebar X does ==="
#!must "success": true
echo-cli call org.geckoOS.evo CloseWindow --interface org.geckoOS.evo --args "{\"window_id\":$WID}"
sleep 12

# Did the process actually exit, or is it lingering with no windows? A
# lingering vela still holds the profile lock, and AppHost still has a live
# session for it — which sends the next Launch down the single-instance
# raise path instead of a fresh spawn.
echo "vela1 still alive: $(kill -0 $VELA1 2>/dev/null && echo YES || echo no)"
echo-cli call org.geckoOS.appHost ListRunningApps --interface org.geckoOS.appHost --args '{}' | grep -cE '"appId": "files"'
ls /var/lib/pyxis/profiles/files/.parentlock 2>&1
ls -la /var/lib/pyxis/profiles/files/sessionstore-backups/ 2>&1 | tail -2

wc -l < /var/log/echo.log > /tmp/mk
echo "=== LAUNCH 2 — the reported failure ==="
echo-cli call org.geckoOS.nova Launch --interface org.geckoOS.nova --args '{"appId":"files"}' | grep -E 'sessionId|Error'
sleep 20

echo "--- what the handler saw on launch 2 ---"
tail -n +$(cat /tmp/mk) /var/log/echo.log | grep -a 'PYXIS-DIAG' | tail -25

echo "--- did a chromeless window get built? ---"
#!must openPyxisAppWindow(): window opened
tail -n +$(cat /tmp/mk) /var/log/echo.log | grep -a 'PYXIS-DIAG'

echo "--- what Evo was handed ---"
grep -a 'ack_configure' /var/log/evo.log | tail -4
# A browser-chrome window is a second toplevel, or one stamped
# vela-default instead of the app id.
#!must-not vela-default
grep -a 'ack_configure' /var/log/evo.log | tail -2

echo "--- processes and sessions ---"
pgrep -af '[-]-pyxis-app files' | head -3
echo-cli call org.geckoOS.appHost ListRunningApps --interface org.geckoOS.appHost --args '{}' | grep -A4 surfaces

# ---------- the actual root cause: Evo never removed destroyed windows ----
# Found 2026-09-21 from a user session's logs. `remove_window_from_workspace`
# was only reachable from the commit handler under `if !root.alive()`, which
# cannot fire for a normally-closed window — destroying a toplevel is when
# commits STOP. XdgShellHandler::toplevel_destroyed was not implemented, and
# the trait's no-op default meant this compiled and silently did nothing.
#
# Symptom in the log: the workspace count only ever rose (1,2,3,4,5,6 over
# three launches of one app) and the removal debug line never appeared once.
# On screen: dead browser-chrome windows stacking over the real app, which
# read as "kiosk mode ignored on relaunch".
#
# Three close/relaunch cycles. The count must come back down, not climb.
echo "=== cycle A ==="
echo-cli call org.geckoOS.nova Launch --interface org.geckoOS.nova --args '{"appId":"terminal"}' >/dev/null; sleep 16
WA=$(echo-cli call org.geckoOS.nova Focus --interface org.geckoOS.nova --args '{"appId":"terminal"}' | grep -o '"windowId": [0-9]*' | grep -o '[0-9]*')
echo-cli call org.geckoOS.evo CloseWindow --interface org.geckoOS.evo --args "{\"window_id\":$WA}" >/dev/null; sleep 10
echo "=== cycle B ==="
echo-cli call org.geckoOS.nova Launch --interface org.geckoOS.nova --args '{"appId":"terminal"}' >/dev/null; sleep 16
WB=$(echo-cli call org.geckoOS.nova Focus --interface org.geckoOS.nova --args '{"appId":"terminal"}' | grep -o '"windowId": [0-9]*' | grep -o '[0-9]*')
echo-cli call org.geckoOS.evo CloseWindow --interface org.geckoOS.evo --args "{\"window_id\":$WB}" >/dev/null; sleep 10
echo "=== cycle C ==="
echo-cli call org.geckoOS.nova Launch --interface org.geckoOS.nova --args '{"appId":"terminal"}' >/dev/null; sleep 18

# Removal must actually happen — this line was absent from an entire session.
#!must Removed window from workspace
grep -a 'Removed window from workspace' /var/log/evo.log | tail -3

# The count is the regression guard. Bloom + one app is 2; with the leak it
# reached 6 over three launches. Anything above 3 means windows are piling up.
#!must WINDOW-COUNT-SANE
test $(grep -a 'total windows:' /var/log/evo.log | sed -E 's/.*total windows: ([0-9]+).*/\1/' | sort -n | tail -1) -le 3 && echo WINDOW-COUNT-SANE || echo WINDOWS-LEAKING
grep -a 'total windows:' /var/log/evo.log | sed -E 's/.*total windows: ([0-9]+).*/count=\1/' | tr '\n' ' '

# And no unstamped leftovers should still be mapped at the end.
#!must ONLY-EXPECTED-WINDOWS
test $(echo-cli call org.geckoOS.appHost ListRunningApps --interface org.geckoOS.appHost --args '{}' | grep -c '"appId"') -eq 1 && echo ONLY-EXPECTED-WINDOWS || echo EXTRA-SESSIONS

# ---------- the actual root cause, found 2026-09-25 --------------------------
# Everything above measures WINDOWS. The bug was never a window: it was the one
# window having its chrome flags rewritten ~2 seconds after it opened, by
# SessionStore.restoreWindowFeatures() reasserting every
# WINDOW_HIDEABLE_FEATURES entry as visible (`aWindow[item].visible =
# !hidden.includes(item)`, with aWinData.hidden undefined). That regenerates
# the root element's `chromehidden` and un-hides #nav-bar -- 40px of URL bar,
# which is what "it looks like a regular Firefox window" was.
#
# Fixed by the SessionStore.sys.mjs patch (gecko/patches/pyxis/...), which
# gives a `pyxisapp` window the same guard upstream already gives its own
# Taskbar Tabs app windows. Read that patch's header before touching this.
#
# Why the assertions below look at `chrome-late` and not at window open: a
# check made at window open PASSES on a broken build. Every earlier probe did,
# which is how this survived two wrong diagnoses. `PYXIS-DIAG chrome-late:`
# (browser-init.js, permanent) logs the settled state 5s in; that is the only
# thing worth asserting.
#
# Three launches of one app, each with a graceful window close in between.
# Launch 1 was ALWAYS clean, on broken builds too -- the restore needs a
# previous session to restore, so the regression only shows from launch 2 on.
# Asserting launch 1 would pass on a build with the bug fully present.
echo "=== chrome-late: three launches, closing the window each time ==="
# The sections above leave `files` running, and nova's single-instance policy
# would raise that window rather than spawn a new one -- no new chrome-late
# line, and every assertion below would pass vacuously.
echo-cli call org.geckoOS.nova Close --interface org.geckoOS.nova --args '{"appId":"files"}' >/dev/null; sleep 8
wc -l < /var/log/echo.log > /tmp/klmk

echo-cli call org.geckoOS.nova Launch --interface org.geckoOS.nova --args '{"appId":"files"}' >/dev/null; sleep 24
K=$(echo-cli call org.geckoOS.nova Focus --interface org.geckoOS.nova --args '{"appId":"files"}' | grep -o '"windowId": [0-9]*' | grep -o '[0-9]*')
echo-cli call org.geckoOS.evo CloseWindow --interface org.geckoOS.evo --args "{\"window_id\":$K}" >/dev/null; sleep 10

echo-cli call org.geckoOS.nova Launch --interface org.geckoOS.nova --args '{"appId":"files"}' >/dev/null; sleep 24
K=$(echo-cli call org.geckoOS.nova Focus --interface org.geckoOS.nova --args '{"appId":"files"}' | grep -o '"windowId": [0-9]*' | grep -o '[0-9]*')
echo-cli call org.geckoOS.evo CloseWindow --interface org.geckoOS.evo --args "{\"window_id\":$K}" >/dev/null; sleep 10

echo-cli call org.geckoOS.nova Launch --interface org.geckoOS.nova --args '{"appId":"files"}' >/dev/null; sleep 24

# The diagnostic has to be there at all, or every assertion below is vacuous.
#!must chrome-late
tail -n +$(cat /tmp/klmk) /var/log/echo.log | grep -a 'chrome-late' | sed 's/.*PYXIS-DIAG/LATE/'

# All three launches must report a chromeless window. toolbarvis is the
# load-bearing one: SessionStore set it true, and TabBarVisibility.update()
# derives the whole popup-like single-tab treatment from it.
#!must-not toolbarvis=true
tail -n +$(cat /tmp/klmk) /var/log/echo.log | grep -a 'chrome-late' | grep -o 'toolbarvis=[a-z]*' | sort | uniq -c

# #nav-bar visible at 40px IS the bug. 0 is the fix. Assert both directions:
# must-not is what catches the regression, must is what catches the case where
# the field stopped being logged at all and must-not passed on an empty grep.
#!must navbar=0
#!must-not navbar=4
tail -n +$(cat /tmp/klmk) /var/log/echo.log | grep -a 'chrome-late' | grep -o 'navbar=[0-9-]*' | sort | uniq -c

# And chromehidden must still carry the kiosk set, not have decayed to
# "extrachrome" (which is what a restored/reasserted window ends up with).
#!must-not chromehidden="extrachrome
tail -n +$(cat /tmp/klmk) /var/log/echo.log | grep -a 'chrome-late' | grep -o 'chromehidden=.*' | sort | uniq -c

# Exactly three launches were made; fewer chrome-late lines means a launch
# never opened a window and every assertion above passed vacuously.
#
# Asserting the bare count, NOT a `&& echo SOME-TOKEN` sentinel: the harness
# strips command echo from the head of the output by matching the flattened
# command text, so a sentinel that appears verbatim in the command that prints
# it gets eaten and the assertion fails on a passing run. Anything under 8
# characters (like "3") is never stripped. Cost an hour on 2026-09-25.
#!must 3
tail -n +$(cat /tmp/klmk) /var/log/echo.log | grep -ac 'chrome-late'
